cmake_minimum_required(VERSION 3.20)
project(photo_hub_sony LANGUAGES CXX)
if(NOT SONY_SDK_ROOT)
  message(FATAL_ERROR "Set SONY_SDK_ROOT to the extracted RemoteCli directory")
endif()
add_executable(photographer-sony main.cpp)
target_compile_features(photographer-sony PRIVATE cxx_std_17)
target_compile_definitions(photographer-sony PRIVATE UNICODE _UNICODE)
target_compile_options(photographer-sony PRIVATE /utf-8)
target_include_directories(photographer-sony PRIVATE "${SONY_SDK_ROOT}/app/CRSDK")
target_link_libraries(photographer-sony PRIVATE "${SONY_SDK_ROOT}/external/crsdk/Cr_Core.lib")
add_custom_command(TARGET photographer-sony POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${SONY_SDK_ROOT}/external/crsdk" "$<TARGET_FILE_DIR:photographer-sony>")
