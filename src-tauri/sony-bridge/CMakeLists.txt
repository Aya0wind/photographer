cmake_minimum_required(VERSION 3.20)
project(photo_hub_sony LANGUAGES CXX)
if(NOT SONY_SDK_ROOT)
  message(FATAL_ERROR "Set SONY_SDK_ROOT to the extracted RemoteCli directory")
endif()
add_executable(photo-hub-sony main.cpp)
target_compile_features(photo-hub-sony PRIVATE cxx_std_17)
target_compile_definitions(photo-hub-sony PRIVATE UNICODE _UNICODE)
target_compile_options(photo-hub-sony PRIVATE /utf-8)
target_include_directories(photo-hub-sony PRIVATE "${SONY_SDK_ROOT}/app/CRSDK")
target_link_libraries(photo-hub-sony PRIVATE "${SONY_SDK_ROOT}/external/crsdk/Cr_Core.lib")
add_custom_command(TARGET photo-hub-sony POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${SONY_SDK_ROOT}/external/crsdk" "$<TARGET_FILE_DIR:photo-hub-sony>")
