cmake_minimum_required(VERSION 3.24)
project(PhotographerRaw LANGUAGES C CXX)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION OFF)
set(JPEGXL_ENABLE_LTO OFF CACHE BOOL "" FORCE)
include(FetchContent)
include(CTest)
# Same source/build revisions as the inspected RawLab core. Download only at
# explicit prototype configure time, never from the desktop build.rs.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/upstream.lock.json" rawlab_lock)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/upstream/.git")
    message(FATAL_ERROR "RawLab submodule missing; run git submodule update --init --recursive")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" rev-parse HEAD
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/upstream"
    RESULT_VARIABLE rawlab_git_status OUTPUT_VARIABLE rawlab_git_commit
    OUTPUT_STRIP_TRAILING_WHITESPACE)
string(JSON rawlab_expected_commit GET "${rawlab_lock}" commit)
if(NOT rawlab_git_status EQUAL 0 OR NOT rawlab_git_commit STREQUAL rawlab_expected_commit)
    message(FATAL_ERROR "RawLab submodule differs from upstream.lock.json; initialize the pinned revision")
endif()
foreach(dependency libraw librawCmake)
    string(JSON ${dependency}_url GET "${rawlab_lock}" nativeDependencies ${dependency} url)
    string(JSON ${dependency}_sha GET "${rawlab_lock}" nativeDependencies ${dependency} sha256)
endforeach()
FetchContent_Declare(pc_libraw_source URL "${libraw_url}" URL_HASH "SHA256=${libraw_sha}")
FetchContent_Declare(pc_libraw_cmake URL "${librawCmake_url}" URL_HASH "SHA256=${librawCmake_sha}"
    SOURCE_SUBDIR photographer-no-autobuild)
FetchContent_MakeAvailable(pc_libraw_source)
set(LIBRAW_SOURCE_DIR "${pc_libraw_source_SOURCE_DIR}")
include(upstream/lutools/cmake/patch-libraw.cmake)
set(LIBRAW_PATH "${LIBRAW_SOURCE_DIR}" CACHE PATH "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
# M0 has no host resource budget yet. Keep internal OpenMP off, including
# X-Trans, instead of oversubscribing future Rust worker pools.
set(ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
set(ENABLE_LCMS OFF CACHE BOOL "" FORCE)
set(ENABLE_JASPER OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_JPEG ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ZLIB REQUIRED) # compressed DNG must not silently lose its decoder
FetchContent_MakeAvailable(pc_libraw_cmake)
option(PHOTOGRAPHER_RAWLAB_JXL "Build upstream JPEG XL DNG support" ON)
if(PHOTOGRAPHER_RAWLAB_JXL)
    set(LIBRAW_CMAKE_SOURCE_DIR "${pc_libraw_cmake_SOURCE_DIR}")
    add_subdirectory(upstream/lutools/cmake/libraw raw-dependencies)
    set(SONY2FUJI_DNGSDK_TARGET rawlab_dng_sdk)
else()
    add_subdirectory("${pc_libraw_cmake_SOURCE_DIR}" raw-dependencies)
endif()
# Upstream source is unchanged; only needed capabilities are configured here.
set(BUILD_TESTING_SAVED "${BUILD_TESTING}")
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SONY2FUJI_LIBRAW_TARGET raw_r)
set(SONY2FUJI_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
option(PHOTOGRAPHER_RAWLAB_GPU "Build upstream platform GPU backends" ON)
set(SONY2FUJI_ENABLE_GPU "${PHOTOGRAPHER_RAWLAB_GPU}" CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_CHROMA_DENOISE OFF CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_WAVELET_DENOISE OFF CACHE BOOL "" FORCE)
set(SONY2FUJI_BUILD_TIFF_DEPS ON CACHE BOOL "" FORCE)
if(WIN32)
    # Upstream checksum-pinned static Lensfun + GLib build.
    set(SONY2FUJI_BUILD_LENS_DEPS ON CACHE BOOL "" FORCE)
elseif(APPLE)
    # Build the same pinned Lensfun against GLib supplied by the host toolchain.
    include(cmake/lensfun-macos.cmake)
elseif(UNIX)
    # Lensfun's pkg-config metadata also needs the GLib development package.
    # Fail at configure time rather than compiling without its include paths.
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(LENSFUN REQUIRED IMPORTED_TARGET GLOBAL lensfun>=0.3.4)
endif()
set(SONY2FUJI_ENABLE_LENSFUN ON CACHE BOOL "" FORCE)
set(SONY2FUJI_ENABLE_DNGSDK "${PHOTOGRAPHER_RAWLAB_JXL}" CACHE BOOL "" FORCE)
add_subdirectory(upstream/lutools)
set(BUILD_TESTING "${BUILD_TESTING_SAVED}" CACHE BOOL "" FORCE)
add_library(photographer_raw SHARED src/photographer_raw.cpp)
target_compile_features(photographer_raw PRIVATE cxx_std_17)
target_compile_definitions(photographer_raw PRIVATE PHOTOGRAPHER_RAW_BUILD)
if(PHOTOGRAPHER_RAWLAB_GPU AND WIN32)
    target_compile_definitions(photographer_raw PRIVATE PC_D3D11)
elseif(PHOTOGRAPHER_RAWLAB_GPU AND APPLE)
    target_compile_definitions(photographer_raw PRIVATE PC_METAL)
elseif(PHOTOGRAPHER_RAWLAB_GPU AND ANDROID)
    target_compile_definitions(photographer_raw PRIVATE PC_GLES)
endif()
target_include_directories(photographer_raw PRIVATE include upstream/lutools/include upstream/lutools/src upstream/lutools/third_party)
target_link_libraries(photographer_raw PRIVATE sony2fuji_core)
if(MSVC)
    target_compile_definitions(photographer_raw PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
    # SDK RGB() expands core constructors to COLORREF. Preserve vendor source.
    set(rgb_compat "${CMAKE_CURRENT_SOURCE_DIR}/include/windows_rgb_compat.h")
    target_compile_options(sony2fuji_core PRIVATE "/FI${rgb_compat}")
    target_compile_options(photographer_raw PRIVATE "/FI${rgb_compat}")
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(raw_abi_test tests/raw_abi_test.cpp)
    target_include_directories(raw_abi_test PRIVATE include)
    target_link_libraries(raw_abi_test PRIVATE photographer_raw)
    add_test(NAME raw_abi COMMAND raw_abi_test)
endif()
